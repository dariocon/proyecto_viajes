selectedTimeFilter: string = ''; allTrips: TripDto[] = []; applyFilters(): void { // Siempre partimos de todos los viajes disponibles let trips = [...this.allTrips]; const now = new Date(); // Filtrado por categoría if (this.category) { trips = trips.filter(t => t.categoryId === this.category); } // Filtrado por tiempo if (this.selectedTimeFilter) { trips = trips.filter(trip => { const startDate = new Date(trip.startDate); switch (this.selectedTimeFilter) { case 'soon': return startDate >= now; case 'far': return startDate >= now; case 'week': const weekFromNow = new Date(); weekFromNow.setDate(now.getDate() + 7); return startDate >= now && startDate <= weekFromNow; case 'month': const monthFromNow = new Date(); monthFromNow.setMonth(now.getMonth() + 1); return startDate >= now && startDate <= monthFromNow; default: return true; } }); // Ordenación if (this.selectedTimeFilter === 'soon' || this.selectedTimeFilter === 'week' || this.selectedTimeFilter === 'month') { trips.sort((a, b) => new Date(a.startDate).getTime() - new Date(b.startDate).getTime()); } else if (this.selectedTimeFilter === 'far') { trips.sort((a, b) => new Date(b.startDate).getTime() - new Date(a.startDate).getTime()); } } this.filteredTrips = trips; this.currentPage = 1; this.applyPagination(); } onTimeFilterChange(): void { this.applyFilters(); }
